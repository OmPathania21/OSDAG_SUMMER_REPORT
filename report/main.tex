%----------------------------------------------------- consistent figure helper
% Usage: \osdagfig{filename}{caption}{label}
% Falls back to a labelled placeholder box when the image is not yet supplied,
% so the report always compiles.
\newcommand{\osdagfig}[3]{%
  \begin{figure}[H]
    \centering
    \IfFileExists{images/#1}{%
      \includegraphics[width=0.85\textwidth]{#1}%
    }{%
      \fbox{\parbox[c][5cm][c]{0.85\textwidth}{\centering\small
        \textbf{IMAGE NEEDED}\\[4pt]\texttt{\detokenize{images/#1}}\\[6pt]\itshape #2}}%
    }
    \caption{#2}
    \label{#3}
  \end{figure}%
}
